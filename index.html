Get started

From zero to production
in a weekend.

quickstart.ts
import { Nexus } from '@nexus/sdk';

const nx = new Nexus({ project: 'my-app' });

const agent = nx.agent({
  tools: ['search', 'email', 'calendar'],
  memory: 'shared',
  policy: './policies/readonly.yaml',
});

await agent.run('Summarize this week and draft my Monday.');
